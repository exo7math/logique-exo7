\documentclass[tikz,border=2pt]{standalone}
\begin{document}
\begin{tikzpicture}[line width=1.5pt, line cap=round, line join=round]

    \def\mylinewidth{0.3}
    \def\trianglesep{2*\mylinewidth}
    \def\R{1.5}
    \def\r{0.75}

    % Triangle gauche
    \draw[rotate=-45, fill] 
        (-0.5,-0.5) 
        [rounded corners=10pt] -- (10,0) 
        [rounded corners=10pt] -- (0,10) 
        [rounded corners=30pt] -- cycle;

    % Triangle droite 
    \begin{scope}[shift={({10*sqrt(2)+\trianglesep},0)}]   
        \draw[rotate=180-45, line width=5pt, fill=white] 
        (-0.5,-0.5) 
        [rounded corners=10pt] -- (10,0) 
        [rounded corners=10pt] -- (0,10) 
        [rounded corners=30pt] -- cycle;
    \end{scope}

    % Letter G
    \begin{scope}[scale=0.7,shift={(4.16,-4.5)}]
        \draw[draw=white, fill=white]
        (0,\R) arc (180:270:\R) -- (\R,0) -- (6,0) -- (6,5) --
        (3,5) -- (3,4) -- (5,4) -- (5,1) -- (1+\r,1) 
        arc (-90:-180:\r) -- (1,1+\r) 
        -- (1,8-\r) 
       arc (180:90:\r) -- (1+\r,8) -- (6,8) -- (6,9) 
      -- (0+\R,9) arc(90:180:\R) --  (0,9-\R) -- cycle;
    \end{scope}

    % Letter G reversed
    \begin{scope}[xscale=-1,yscale=-1,scale=0.7,shift = {(-16.87,-4.5)}]
        \draw[draw=white,fill=white] (4.,4.1) rectangle (6.1,8.5);
        \draw[draw=black, fill=black]
        (0,\R) arc (180:270:\R) -- (\R,0) -- (6,0) -- (6,5) --
        (3,5) -- (3,4) -- (5,4) -- (5,1) -- (1+\r,1) 
        arc (-90:-180:\r) -- (1,1+\r) 
        -- (1,8-\r) 
       arc (180:90:\r) -- (1+\r,8) -- (6,8) -- (6,9) 
      -- (0+\R,9) arc(90:180:\R) --  (0,9-\R) -- cycle;
    \end{scope}

\end{tikzpicture}
\end{document}
